\newglossaryentry{ab initio}
{
    name=ab initio,
    description={Latin term meaning \textit{from the beginning}; in computational
        physics and chemistry, refers to calculations derived solely from
        fundamental physical principles, without empirical parameters or
        experimental fitting}
}
